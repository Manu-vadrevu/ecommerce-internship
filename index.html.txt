        <button style="width: 100%;" onclick="login()">Enter Store</button>

        <div style="overflow: auto; margin-bottom: 1.5rem;">
            <button class="logout-btn" onclick="logout()">Logout</button>

            <h3>Admin Panel: Add New Catalog Product</h3>
            <div style="display: flex; gap: 10px;">

                <input type="number" id="prodPrice" placeholder="Price (INR)">

                <button class="checkout-btn" onclick="checkoutOrder()">Place Secure Order</button>

<script>
    let currentRole = "User";
    let currentUsername = "";
    let cart = [];

    function login() {
        const user = document.getElementById('usernameInput').value.trim();
        if(!user) return alert("Please enter a username");
        
        currentUsername = user;
        currentRole = document.getElementById('roleInput').value;
        
        document.getElementById('userBadge').innerText = currentUsername;
        document.getElementById('roleBadge').innerText = currentRole;
        
        document.getElementById('loginScreen').classList.add('hidden');
        document.getElementById('storeScreen').classList.remove('hidden');
        
        if(currentRole === "Admin") {
            document.getElementById('adminPanel').classList.remove('hidden');
        } else {
            document.getElementById('adminPanel').classList.add('hidden');
        }
        
        loadProducts();
    }

    function logout() {
        cart = [];
        document.getElementById('cartCount').innerText = "0";
        document.getElementById('loginScreen').classList.remove('hidden');
        document.getElementById('storeScreen').classList.add('hidden');
    }

    async function loadProducts() {
        const res = await fetch('/api/products');
        const products = await res.json();
        const container = document.getElementById('productsContainer');
        container.innerHTML = '';
        
        products.forEach(p => {
            const div = document.createElement('div');
            div.className = 'product-card';
            div.innerHTML = `
                <h3>${p.title}</h3>
                <div class="product-price">₹${p.price}</div>
                <button onclick="addToCart('${p.title}')">Add to Cart</button>
            `;
            container.appendChild(div);
        });
    }

    async function createProduct() {
        const title = document.getElementById('prodName').value.trim();
        const price = document.getElementById('prodPrice').value.trim();
        if(!title || !price) return alert("Fill in product details");

        await fetch('/api/products', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ title, price: parseInt(price) })
        });

        document.getElementById('prodName').value = '';
        document.getElementById('prodPrice').value = '';
        loadProducts();
    }

    function addToCart(title) {
        cart.push(title);
        document.getElementById('cartCount').innerText = cart.length;
    }

    function checkoutOrder() {
        if(cart.length === 0) return alert("Your cart is empty!");
        alert(`🎉 Order successful! Thank you for buying ${cart.length} items.`);
        cart = [];
        document.getElementById('cartCount').innerText = "0";
    }
</script>
